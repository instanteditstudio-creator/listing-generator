function buildPrompt(vibe,themeKey,sceneIndex){ var theme=THEMES[themeKey]||THEMES.pool_party; var kw=interpretKeywords(vibe); var idx=typeof sceneIndex==='number'?sceneIndex%3:0; var angles=['overhead flat lay, birds-eye view', '45-degree angled product shot, diagonal composition', 'close-up detail, shallow depth of field, soft bokeh']; var parts=[]; // YOUR keywords lead — highest token weight if(kw.props.length) parts.push(kw.props.join(', ')); if(kw.colors.length) parts.push(kw.colors.join(', ')); if(kw.mood) parts.push(kw.mood); // theme scene supports, doesn't dominate parts.push(theme.scenes[idx]); parts.push(angles[idx]); parts.push('photorealistic styled product photography, no text, no people'); return {main:parts.join(', ')}; } async function getAIBg(promptObj){ var p=typeof promptObj==='string'?promptObj:promptObj.main; for(var attempt=0;attempt<2;attempt++){ try{ var seed=Math.floor(Math.random()*99999); var url='https://image.pollinations.ai/prompt/'+encodeURIComponent(p) +'?width=1024&height=1024&seed='+seed+'&nologo=true'; var res=await fetch(url); if(!res.ok){await sleep(1500);continue;} var blob=await res.blob(); if(blob.size<5000||blob.type.indexOf('image')===-1){await sleep(1500);continue;} return await new Promise(function(r){var fr=new FileReader();fr.onload=function(){r(fr.result);};fr.onerror=function(){r(null);};fr.readAsDataURL(blob);}); }catch(e){await sleep(1500);} } console.warn('AI background failed for prompt:',p); return null; }
